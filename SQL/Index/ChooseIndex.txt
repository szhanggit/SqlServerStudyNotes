INNER JOIN dbo.ClientQuotationProductSoldPrice CQPSP WITH (NOLOCK, INDEX = [IDX_ClientQuotationProductSoldPrice_Validity]) ON  CQPSP.[ValidityFrom] <= OBI.[ActiveDate]
                                                                                                                                                        AND CQPSP.[ValidityEnd] >= OBI.[ActiveDate]
                                                                                                                                                        AND CQPSP.STATUS = 2
