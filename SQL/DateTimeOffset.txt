    SYSDATETIMEOFFSET() AS 'Current Date',
    SWITCHOFFSET( SYSDATETIMEOFFSET(), '+09:00' ) AS '+09:00';

Convert local datetime to be utc under the current database setting.

Declare
    @OrderDate AS DATETIME,
    @OrderDateUTC As DATETIME;
 
SET @OrderDate = '2021-08-15 18:32:15';
SET @OrderDateUTC = DATEADD(second, 
    DATEDIFF(second, GETDATE(), GETUTCDATE()), @OrderDate);

select @OrderDateUTC as OrderDateUTC
