USE AdventureWorks2014

-- Basic example of TRY...CATCH
 
BEGIN TRY
-- Generate a divide-by-zero error  
	SELECT
		1 / 0 AS Error;
END TRY
BEGIN CATCH
	SELECT
		ERROR_NUMBER() AS ErrorNumber,
		ERROR_STATE() AS ErrorState,
		ERROR_SEVERITY() AS ErrorSeverity,
		ERROR_PROCEDURE() AS ErrorProcedure,
		ERROR_LINE() AS ErrorLine,
		ERROR_MESSAGE() AS ErrorMessage;
END CATCH;

ALTER PROCEDURE [dbo].[spInsertTransaction] (
	@ErrorMessage NVARCHAR(4000) OUTPUT
	,@ErrorLine INT OUTPUT
	)
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

	BEGIN TRAN

	BEGIN TRY


	END TRY

	BEGIN CATCH
		SELECT @ErrorMessage = ERROR_MESSAGE()

		SELECT @ErrorLine = ERROR_LINE()

		IF @@TRANCOUNT > 0
			ROLLBACK TRANSACTION;
	END CATCH;

	IF @@TRANCOUNT > 0
		COMMIT TRANSACTION;
END

ALTER PROCEDURE [dbo].[spUpdateVoucherConsumerInfo]
	@IsSuccess                BIT		OUTPUT
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON; 
	
    BEGIN TRANSACTION;
	
	SET @IsSuccess = 0;

	BEGIN TRY

		UPDATE Voucher
		SET ActivatedOn = VSB.ActivatedOn
			,BalanceAvailable = VSB.BalanceAvailable
			,BalanceFrozen = VSB.BalanceFrozen
			,ExpiryDate = IIF(VSB.Status = 256 OR VSB.Status = 8, V.ExpiryDate,  VSB.ExpiryDate)
			,PublishedOn = VSB.PublishedOn
			,[Status] = VSB.[Status]
			,ConsumeTime = VSB.ConsumeTime
			,ConsumeMerchantCode = VSB.ConsumeMerchantCode
			,ConsumeShopCode = VSB.ConsumeShopCode
			,ConsumeTerminalSSN = VSB.ConsumeTerminalSSN
			,LastTranOn = VSB.LastTranOn
		FROM VoucherSyncBuffer AS VSB
			INNER JOIN Voucher AS V ON VSB.VoucherNumber = V.VoucherNumber
			INNER JOIN Program AS P ON VSB.ProgramCode = P.IdentityCode AND V.ProgramId = P.Id
				
	END TRY

	BEGIN CATCH

		IF @@TRANCOUNT > 0
				ROLLBACK TRANSACTION;
		RETURN
	END CATCH	

	IF @@TRANCOUNT > 0
		BEGIN
			COMMIT TRANSACTION;
			SET @IsSuccess = 1;
		END
END
