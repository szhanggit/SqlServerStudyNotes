ALTER FUNCTION [dbo].[func_Split] 
    (   
    @DelimitedString    varchar(8000),
    @Delimiter              varchar(100) 
    )
RETURNS @tblArray TABLE
    (
    Element     varchar(1000)               -- Array element contents
    )
AS
BEGIN

    -- Local Variable Declarations
    -- ---------------------------
    DECLARE @Index      smallint,
                    @Start      smallint,
                    @DelSize    smallint

    SET @DelSize = LEN(@Delimiter)

    -- Loop through source string and add elements to destination table array
    -- ----------------------------------------------------------------------
    WHILE LEN(@DelimitedString) > 0
    BEGIN

        SET @Index = CHARINDEX(@Delimiter, @DelimitedString)

        IF @Index = 0
            BEGIN

                INSERT INTO
                    @tblArray 
                    (Element)
                VALUES
                    (LTRIM(RTRIM(@DelimitedString)))

                BREAK
            END
        ELSE
            BEGIN

                INSERT INTO
                    @tblArray 
                    (Element)
                VALUES
                    (LTRIM(RTRIM(SUBSTRING(@DelimitedString, 1,@Index - 1))))

                SET @Start = @Index + @DelSize
                SET @DelimitedString = SUBSTRING(@DelimitedString, @Start , LEN(@DelimitedString) - @Start + 1)

            END
    END

    RETURN
END

ALTER FUNCTION [dbo].[F_SplitString]
(
	@SourceString VARCHAR(MAX),
	@Seperator VARCHAR(25) = ','
)
RETURNS @ResultTable TABLE
(
	[Position] INT IDENTITY(1,1),
	[Value] VARCHAR(MAX)
)
AS
BEGIN
    DECLARE @w_xml xml;

    SET @w_xml = N'<root><i>' + replace(@SourceString, @Seperator,'</i><i>') + '</i></root>';

    INSERT INTO @ResultTable ([Value])
    SELECT [i].value('.', 'VARCHAR(MAX)') AS Value
    FROM @w_xml.nodes('//root/i') AS [Items]([i]);
    RETURN;
    END;

create PROCEDURE [dbo].[spScanApiLog]
    @StartIndex int,
    @EndIndex int
AS
    BEGIN

        SET NOCOUNT ON;

		declare @DateTimeMark DateTime = (select top(1) CreatedOn from ApiLog with(nolock) where Id = @EndIndex)
		declare @newcreatedindex int;


			insert into ApiLogScanLog20210918 (StartIndex, EndIndex, DateTimeMark, CreatedDate) values (@StartIndex, @EndIndex, @DateTimeMark, getdate());
			set @newcreatedindex = SCOPE_IDENTITY();




			insert into ApiLogScan20210918 (ApiLogId, ConsumerCode, OperationCode, ExpiryDate, VoucherNumber, ProgramCode, VoucherGuid)
			select Id as ApiLogId, ConsumerCode, OperationCode
			, substring([ResponseMessage], CHARINDEX('&lt;ExpirationDate&gt;', [ResponseMessage], 0) + 22, 19) as ExpiryDate 
			, substring([ResponseMessage], CHARINDEX('&lt;VoucherNo&gt;', [ResponseMessage], 0) + 17, CHARINDEX('&lt;/VoucherNo&gt;', [ResponseMessage], 0)-CHARINDEX('&lt;VoucherNo&gt;', [ResponseMessage], 0)-17) as VoucherNumber
			, substring([ResponseMessage], CHARINDEX('&lt;ProgramCode&gt;', [ResponseMessage], 0) + 19, CHARINDEX('&lt;/ProgramCode&gt;', [ResponseMessage], 0)-CHARINDEX('&lt;ProgramCode&gt;', [ResponseMessage], 0)-19) as ProgramCode
			, null as VoucherGuid
			from ApiLog with(nolock)
			where Id >= @StartIndex AND Id <=  @EndIndex AND OperationCode in (
			'GetVoucherV2',
			'GetVoucherV3',
			'GetVoucherV5',
			'GetVoucher',
			'RequestVouchersIdempotent'
			) AND ResponseCode = '0000' AND ResponseMessage is not null AND ResponseMessage != ''
			order by Id desc




			insert into ApiLogScan20210918 (ApiLogId, ConsumerCode, OperationCode, ExpiryDate, VoucherNumber, ProgramCode, VoucherGuid)
			select Id as ApiLogId, ConsumerCode, OperationCode
			, substring([ResponseMessage], CHARINDEX('&lt;ExpirationDate&gt;', [ResponseMessage], 0) + 22, 19) as ExpiryDate 
			, (select top(1) Element from dbo.func_split(substring([ResponseMessage], CHARINDEX('&lt;VoucherNo&gt;', [ResponseMessage], 0) + 17, CHARINDEX('&lt;/VoucherNo&gt;', [ResponseMessage], 0)-CHARINDEX('&lt;VoucherNo&gt;', [ResponseMessage], 0)-17), '|')) as VoucherNumber
			, substring([ResponseMessage], CHARINDEX('&lt;ProgramCode&gt;', [ResponseMessage], 0) + 19, CHARINDEX('&lt;/ProgramCode&gt;', [ResponseMessage], 0)-CHARINDEX('&lt;ProgramCode&gt;', [ResponseMessage], 0)-19) as ProgramCode
			, null as VoucherGuid
			from ApiLog with(nolock)
			where Id >= @StartIndex AND Id <=  @EndIndex AND OperationCode in (
			'GetMultiVouchers',
			'GetMultiVouchersV2',
			'GetMultiVouchersV3',
			'GetMultiVouchersV6',
			'SendMultiVouchers'
			) AND ResponseCode = '0000' AND ResponseMessage is not null AND ResponseMessage != ''
			order by Id desc



			insert into ApiLogScan20210918 (ApiLogId, ConsumerCode, OperationCode, ExpiryDate, VoucherNumber, ProgramCode, VoucherGuid)
			select Id as ApiLogId, ConsumerCode, OperationCode
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ResendVoucherResponse/ExpirationDate/text()') as varchar(100)) as ExpiryDate
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ResendVoucherResponse/VoucherNo/text()') as varchar(100)) as VoucherNumber
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ResendVoucherResponse/ProgramCode/text()') as varchar(100)) as ProgramCode
			, null as VoucherGuid
			from ApiLog with(nolock)
			where Id >= @StartIndex AND Id <=  @EndIndex AND ResponseCode = '0000' AND OperationCode in (
			'ResendVoucher',
			'SendVoucher'
			) AND ResponseMessage is not null AND ResponseMessage != ''
			order by Id desc






			insert into ApiLogScan20210918 (ApiLogId, ConsumerCode, OperationCode, ExpiryDate, VoucherNumber, ProgramCode, VoucherGuid)
			select top(1) Id, ConsumerCode, OperationCode	
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ConfirmChildProductResponse/ExpirationDate/text()') as varchar(100)) as ExpiryDate	
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ConfirmChildProductResponse/VoucherNo/text()') as varchar(100)) as VoucherNumber	
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/ConfirmChildProductResponse/ProgramCode/text()') as varchar(100)) as ProgramCode	
			, null as VoucherGuid
			from ApiLog with(nolock)	
			where Id >= @StartIndex AND Id <=  @EndIndex AND ResponseCode = '0000' AND OperationCode = 'ConfirmChildProduct' AND ResponseMessage is not null AND ResponseMessage != ''	
			order by Id desc	
	
	

			insert into ApiLogScan20210918 (ApiLogId, ConsumerCode, OperationCode, ExpiryDate, VoucherNumber, ProgramCode, VoucherGuid)
			select top(1) Id, ConsumerCode, OperationCode	
			, cast(CAST(replace([ResponseMessage], '<?xml version="1.0" encoding="utf-8"?>', '') AS XML).query('/GetComboChildByVoucherGuidResponse/Childs/ChildOfMasterVoucherInfo/ExpiryDate/text()') as varchar(100)) as ExpiryDate

			where Id >= @StartIndex AND Id <=  @EndIndex AND ResponseCode = '0000' AND OperationCode = 'GetComboChildByVoucherGuid' AND ResponseMessage is not null AND ResponseMessage != ''	
			order by Id desc	









			update ApiLogScanLog20210918

			where Id = @newcreatedindex        

		
    END
