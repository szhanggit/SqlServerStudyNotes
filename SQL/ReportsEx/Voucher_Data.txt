select 
v.VoucherNumber as [Column A -- Voucher Number],
PD.ProductCode as [Column B -- Product Code],
PD.ProductName as [Column C -- Product Name],
CASE
	WHEN PD.IsMasterProduct = 1 THEN 'Master'
    WHEN VC.ChildVoucherId IS NULL THEN 'General'
    ELSE 'Child'
END AS [Column D -- Selection Type],
CASE
	WHEN PD.ProductIssuerType = 0 THEN 'Issuer'
	WHEN PD.ProductIssuerType = 2 THEN 'Reseller'
	WHEN PD.ProductIssuerType = 3 THEN 'White'
	ELSE 'Unknown'
END as [Column E -- Product Issuer],
CASE 
	WHEN (PD.ProductType = 1 AND PD.ProductSubType = 1) THEN 'One Time'
	ELSE 'Multiple Time'
END as [Column F -- Redemption Time],
CASE 
	WHEN PD.ProductType = 1 THEN 
	(
		Case 
			WHEN PD.ProductSubType = 1 THEN 'ProductBased'
			WHEN PD.ProductSubType = 3 THEN 'SmartBooklet'
			WHEN PD.ProductSubType = 5 THEN 'SmartChoiceVoucher'
			ELSE 'Error'
		END
	)
	WHEN PD.ProductType = 2 THEN 
	(
		Case 
			WHEN PD.ProductSubType = 2 THEN 'ValueBased'
			WHEN PD.ProductSubType = 4 THEN 'DynamicFaceValue'
			WHEN PD.ProductSubType = 6 THEN 'ValueBasedOneTimeRedeem'
			WHEN PD.ProductSubType = 8 THEN 'FixedSuperVoucher'
			WHEN PD.ProductSubType = 10 THEN 'DynamicSuperVoucher'
			ELSE 'Error'
		END	
	)
	ELSE 'Error'
END as [Column G -- Product Type],
CASE 
	WHEN PD.VoucherNumberGenerateWay = 1 THEN 'Edenred'
	WHEN PD.VoucherNumberGenerateWay = 2 THEN 'Third Party'
END as [Column H -- Generated By],
OBI.Email as [Column I -- Beneficiery Email],
OBI.Mobile as [Column J -- Beneficiary Mobile Number],
CASE 
	WHEN ProductType = 1 THEN PPPB.FaceValue
	WHEN ProductType = 2 THEN IIF(PD.ProductSubType = 4, OBI.FaceValue, PPVB.FaceValue)
	ELSE 0 
END AS [Column K -- Market Price],
CASE
	WHEN ProductType = 1 THEN CQPSP.SoldPriceWithoutTax
	WHEN ProductType = 2 THEN IIF(PD.ProductSubType = 4, OBI.SoldPrice, ISNULL(CQPSP.ValueBasedSoldPriceWithoutTax, CQPSP.SoldPriceWithoutTax))
	ELSE 0
END AS [Column L -- Default Selling Price],
CASE 
	WHEN ProductType = 1 THEN PPPB.Cost
	WHEN ProductType = 2 THEN IIF(PD.ProductSubType = 4, PPDFV.DefaultProductCost, PPVB.MerchantFee)
	ELSE 0.0
END AS [Column M -- Default Product Cost],
TT.BalanceAvailable as [Column N -- Balance],
TT.TranType as [Column O -- Transaction Type],
TT.ResponseCode as [Column P -- Transaction Status],
CASE
	WHEN v.[Status] = 1 THEN 'Issued'
	WHEN v.[Status] = 2 THEN 'Active'
	WHEN v.[Status] = 4 THEN 'Expired'
	WHEN v.[Status] = 8 THEN 'Trashed'
	WHEN v.[Status] = 16 THEN 'Blocked'
	WHEN v.[Status] = 32 THEN 'Used'
	WHEN v.[Status] = 64 THEN 'InStock'
	ELSE 'error'
END	as [Column Q -- Voucher Status],
TT.TranLocalDateTime as [Column R -- Transaction Date],
v.ExpiryDate as [Column S -- Expiry Date],
o.OrderNumber as [Column T -- TX Order Number],
cq.QuotationNumber as [Column U -- Quotation Number],
CASE
	WHEN ProductType = 1 THEN CQPSP.SoldPriceWithoutTax
	WHEN ProductType = 2 THEN IIF(PD.ProductSubType = 4, OBI.SoldPrice, ISNULL(CQPSP.ValueBasedSoldPriceWithoutTax, CQPSP.SoldPriceWithoutTax))
	ELSE 0
END AS [Column V -- Quotation Sold Price],
C.IdentityCode as [Column W -- Client Code],
C.[Name] as [Column X -- Client Name],
TT.MerchantCode as [Column Y -- Product Merchant Code],
Case
	WHEN TT.MerchantCode = '000000000000000' THEN 'Internal'
	ELSE M.[name] 
END as [Column Z -- Product Merchant Name]
FROM dbo.[Transaction] TT with(nolock)
                          LEFT JOIN dbo.Merchant M WITH(NOLOCK) ON TT.MerchantCode = M.IdentityCode
                          LEFT JOIN dbo.Shop S WITH(NOLOCK) ON TT.ShopCode = S.IdentityCode
                          INNER JOIN dbo.Program PG WITH(NOLOCK) ON TT.ProgramCode = PG.IdentityCode
                          INNER JOIN dbo.Voucher V WITH(NOLOCK) ON TT.AccountNumber = V.VoucherNumber
                                                                   AND PG.Id = V.ProgramId
                          LEFT JOIN dbo.VoucherCombo VC WITH(NOLOCK) ON VC.ChildVoucherId = V.Id --change
                                                                        AND VC.STATUS = 1 --change
                          INNER JOIN dbo.OrderBeneficiaryInfo OBI WITH(NOLOCK) ON V.BeneficiaryInfoId = OBI.Id
						  INNER JOIN dbo.OrderLine OL WITH(NOLOCK) ON OBI.OrderLineId = OL.Id
                          INNER JOIN dbo.[Order] O with(nolock) on OL.OrderId = O.Id
						  INNER JOIN dbo.Dictionary D WITH(NOLOCK) ON TT.TranType = D.NAME
                                                                      AND D.Category = 'TransactionType'
                          LEFT JOIN dbo.ClientQuotationPricing CQPC WITH (NOLOCK, INDEX = [IDX_ClientQuotationPricing_Validity]) ON OL.ClientQuotationProductId = CQPC.ClientQuotationProductId
                                                                                                                                    AND D.Id = CQPC.TransactionTypeId
                                                                                                                                    AND CQPC.[ValidityFrom] <= OBI.[ActiveDate]
                                                                                                                                    AND CQPC.[ValidityEnd] >= OBI.[ActiveDate]
                          INNER JOIN dbo.ClientQuotationProduct CQPD WITH(NOLOCK) ON CQPD.Id = OL.ClientQuotationProductId
                          INNER JOIN dbo.ClientQuotationProductSoldPrice CQPSP WITH (NOLOCK, INDEX = [IDX_ClientQuotationProductSoldPrice_Validity]) ON CQPD.Id = CQPSP.ClientQuotationProductId
                                                                                                                                                        AND CQPSP.[ValidityFrom] <= OBI.[ActiveDate]
                                                                                                                                                        AND CQPSP.[ValidityEnd] >= OBI.[ActiveDate]
                                                                                                                                                        AND CQPSP.STATUS = 2
                          INNER JOIN dbo.ClientQuotation CQ WITH(NOLOCK) ON CQ.Id = CQPD.ClientQuotationId
						  INNER JOIN dbo.Client C with(nolock) ON CQ.ClientId = C.Id
                          INNER JOIN dbo.BusinessModel BM WITH(NOLOCK) ON CQ.BusinessModelId = BM.Id
                          LEFT JOIN dbo.BusinessModelTransactionType BMTT WITH(NOLOCK) ON BMTT.TransactionTypeId = D.Id
                                                                                          AND BMTT.BusinessModelId = CQ.BusinessModelId
                          INNER JOIN dbo.ProductVersion PV WITH(NOLOCK) ON OBI.ProductVersionId = PV.Id
                          INNER JOIN dbo.Product PD WITH(NOLOCK) ON PV.ProductId = PD.Id
                          LEFT JOIN dbo.ProductPricingProductBased PPPB WITH(NOLOCK) ON PV.ProductPricingProductBasedId = PPPB.Id
                          LEFT JOIN dbo.ProductPricingValueBased PPVB WITH(NOLOCK) ON PV.ProductPricingValueBasedId = PPVB.Id
                          LEFT JOIN dbo.ProductPricingDynamicFaceValue PPDFV WITH(NOLOCK) ON PV.Id = PPDFV.ProductVersionId
						  where TT.AccountNumber = 'EEEX2WGXZNC241YPF75FLRZ7AWX97QG27'

+'/'+CAST(DATEPART(mi, GETDATE()) as varchar(10))
+'/'+CAST(DATEPART(ss, GETDATE()) as varchar(10))
