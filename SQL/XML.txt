select top 8 alog.RequestMessage
, cast(alog.RequestMessage as XML).value('(/*:RequestDto/*:OperationCode)[1]', 'varchar(255)') as OperationCode 
, case 
	when cast(alog.RequestMessage as XML).value('(/*:RequestDto/*:OperationCode)[1]', 'varchar(255)') = 'SendMultiVouchersV4' then cast(cast(alog.RequestMessage as XML).value('(/*:RequestDto/*:Parameters)[1]', 'nvarchar(max)') as XML).value('(/*:SendMultiVouchersRequestDto/*:OrderNumber)[1]', 'varchar(255)')
	when cast(alog.RequestMessage as XML).value('(/*:RequestDto/*:OperationCode)[1]', 'varchar(255)') = 'SendMultiVoucherv7' then cast(cast(alog.RequestMessage as XML).value('(/*:RequestDto/*:Parameters)[1]', 'nvarchar(max)') as XML).value('(/*:SendMultiVoucherV7RequestDto/*:OrderNumber)[1]', 'varchar(255)')
end as OrderNumber 
from APILog alog with(nolock)
where alog.RequestMessage != '' AND alog.RequestMessage like '%RequestDto%'
