DECLARE @dateStart datetime
SET @dateStart = CONVERT(nvarchar(10), DATEADD(Hour, -12, GETDATE()), 111)
SET @dateStart = CONVERT(NVARCHAR(10), @dateStart, 111)

DECLARE @dateEnd datetime
SET @dateEnd = DATEADD(DAY, 1, @dateStart)

--ADORA--

select TranLocalDateTime, sum(qty) as qty from 
(
select convert(nvarchar(13), t.TranLocalDateTime, 120) + ':00' as TranLocalDateTime, COUNT(1) qty
from TW_EV_DBS_ADORA.DBO.[Transaction] t WITH(NOLOCK)
where t.TranLocalDateTime >= @dateStart AND t.TranLocalDateTime < @dateEnd
and t.TranType <> 504
and t.ResponseCode = '0000'
group BY convert(nvarchar(13), t.TranLocalDateTime, 120)

UNION

select convert(nvarchar(13), t.TranLocalDateTime, 120) + ':00' as TranLocalDateTime, COUNT(1) qty
from TW_EV_DBS_ADORA.DBO.TransactionHistory t WITH(NOLOCK)
where t.TranLocalDateTime>= @dateStart AND t.TranLocalDateTime < @dateEnd
and t.TranType <> 504
and t.ResponseCode = '0000'
group BY convert(nvarchar(13), t.TranLocalDateTime, 120)

) a
group by TranLocalDateTime
order by TranLocalDateTime Desc
